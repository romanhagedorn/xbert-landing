async function routeInboundCall(session) {
  const intent = await detectIntent(session.transcript);
  if (intent.confidence < 0.82) {
    return escalateToHuman(session, { context: intent.summary });
  }
  switch (intent.type) {
    case "book_appointment":
      return calendar.book(session.callerId, intent.slots);
    case "faq":
      return knowledgeBase.answer(intent.query);
  }
}
